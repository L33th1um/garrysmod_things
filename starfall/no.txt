hook.add("tick","no",function() 
    local e = find.inSphere(owner():getPos(),5000, function(ent) return ent:isPlayer() && ent != owner() end)
    local p = find.closest(e,owner():getPos())
    //for i,p in ipairs(pls) do
        if p == owner() then return end
        p.tl = find.inCone(p:getEyePos(),p:getForward(),5000,0.866, function(ent)
            return ent != p.b && ent == owner()
        end)
        //print(p)
        p.vic = find.closest(p.tl,p:getEyePos())
        if p.vic == nil then return end
        if(p.vic:isValid()) then
            //print(p)
            if(p.bl == nil || !p.bl:isValid()) then
                p.bl = prop.create(p:getEyePos(),p:getEyeAngles() - Angle(90,0,0),"models/hunter/plates/plate075x075.mdl", true)
                p.bl:setNocollideAll(true)
                p.bl:setColor(Color(0,0,0,0))
            else
                p.bl:setPos(p:getEyePos()+ p:worldToLocalVector(p:getForward()*20))
                p.bl:setAngles(p:getEyeAngles() - Angle(90,0,0))
            end 
        end
    //end
            
end)
