local switch = false
hook.add("think", "nigga", function ()
    local ls = find.byModel("models/props_foliage/shrub_small.mdl")
    for i,v in ipairs(ls) do
        if(v:getOwner() == owner()) then
            local scale = - 3/v.c + 4
            local alpha = -math.sqrt(math.exp(v.c-600)) + 100
            v:setScale(Vector(scale*2,scale*2,scale/4))
            v:setColor(Color(0,100,0,alpha))
            v.screen:setScale(Vector(scale*8,scale*8,scale*8))
            v.floor:setScale(Vector(scale*8,scale*8,scale/10))
            v.up:setScale(Vector(scale*8,scale*8,scale/10))
            v.c = v.c + 1 
            if alpha < 0 then v:remove() end
        end    
    end
end)

hook.add("KeyPress", "smoke", function(ply, key) 
    if ply != owner() then return end
     if(key == IN_KEY.ATTACK2 && owner():getActiveWeapon():getClass() == "weapon_crowbar") then
        local frag = prop.create(owner():getEyePos(), owner():getEyeAngles(), "models/Items/grenadeAmmo.mdl")
        frag:emitSound("weapons/grenade/tick1.wav")
        frag:setTrails(32,0,5,"trails/smoke", Color(0,200,0))
        frag:getPhysicsObject():setVelocity(frag:getForward() * 500 * frag:getMass() + Vector(0,0,50) + owner():getVelocity())
        timer.simple(2, function()
            local holo = holograms.create(frag:getPos(), Angle(), "models/props_foliage/shrub_small.mdl")
            holo:setMaterial("effects/fake_fog01b")
            holo:emitSound("weapons/smokegrenade/smoke_emit.wav")
            holo.c = 0
            holo.screen = holograms.create(holo:getPos(), Angle(0,0,0), "models/holograms/hq_stube_thin.mdl")
            holo.screen:setMaterial("shadertest/unlitsinglecolor")
            holo.screen:setParent(holo)
            holo.floor = holograms.create(holo:getPos(), Angle(0,0,0), "models/holograms/cube.mdl")
            holo.floor:setParent(holo)
            holo.floor:setMaterial("shadertest/unlitsinglecolor")
            holo.floor:setColor(Color(0,0,0,255))
            
            holo.up = holograms.create(holo:getPos() + Vector(0,0,180), Angle(0,0,0), "models/holograms/cube.mdl")
            holo.up:setParent(holo)
            holo.up:setMaterial("shadertest/unlitsinglecolor")
            holo.up:setColor(Color(0,0,0,255))
            //holo:setRenderMode(RENDERMODE.WORLDGLOW)
            holo.screen:setColor(Color(0,0,0,255))
            holo.screen:setRenderMode(9)
            frag:remove()
        end)
    end
    if(key == IN_KEY.WALK) then
        if(switch == false) then
            door = hologram.create(owner():getPos() + Vector(0,0,45) + owner():worldToLocalVector(owner():getForward()*50), Angle(0,owner():getEyeAngles().yaw,0)-Angle(90,0,0), "models/props_phx/wheelaxis.mdl", Vector(5,5,1))        
            door:setParent(owner())
            door:setMaterial("models/props_combine/tprings_globe")  
            
            //print("niggaq")
            switch = true  
        else
            door:remove()
            switch = false
        end
    end
end)
